        'Elige un numero de la lista o escribe el puesto que buscas:\n\n' + formatJobTitlesList(),
        attempts,

    data.puesto_buscado = puesto;

        `Responde con una opcion valida:

2️ 1 a 2 anos
3️ 3 a 5 anos
4️ Mas de 5 anos`,

      break;

    data.experiencia = experiencia;

1️⃣ Inmediatamente

4️⃣ En un mes`;
    nextStep = 'esperando_disponibilidad';
    break;
  }

  case 'esperando_disponibilidad': {
    const disponibilidad = parseDisponibilidad(normalized);

    if (!disponibilidad) {
      reply = `Responde con una opción válida:

1️⃣ Inmediatamente
2️⃣ Esta semana
3️⃣ En 15 días
4️⃣ En un mes`;
      break;

    reply = `¿Qué horario prefieres?

🌞 Matutino

1️⃣ Menos de $10000
2️⃣ $10000 a $15000

4️⃣ Más de $20000`;

    if (!documentacion) {
      reply = `Responde con una opción válida:

✅ Si completa
🟡 Me faltan algunos documentos
❌ No la tengo completa`;
      break;

    data.documentacion = documentacion;

      nextStep = 'confirmacion_resumen';
      break;
    }

    const curp = text.toUpperCase().replace(/\s/g, '');

    if (!isValidCurp(curp)) {

        attempts >= 2
          ? 'La CURP no es valida. Puedes escribir "continuar sin CURP" para seguir, o intenta de nuevo en mayusculas.'

        attempts,

    reply = buildSummary(data);

  case 'confirmacion_resumen':
    if (isYes(normalized)) {

      if (pending && pending.id) {
        selectedVacancy = pending;

        data.selected_vacancy = {

          title: pending.title || '',
          company_name: pending.company_name || '',
          location: pending.location || '',

        shouldUpsertCandidate = true;

      shouldSearchVacancies = true;
